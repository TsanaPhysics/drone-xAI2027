ตัวอย่างโค้ดคำนวณ VARI Index ในภาษา Dart: 
double computeVARI(int r, int g, int b) {
  // VARI = (Green - Red) / (Green + Red - Blue)
  double num = (g - r).toDouble();
  double den = (g + r - b).toDouble();
  if (den == 0) return 0.0;
  return (num / den).clamp(-1.0, 1.0);
}
